MEDIA MARKT 

1) Créez une base de données « mediamarkt » utf8mb4 en generale_ci.

CREATE DATABASE mediamarkt 
DEFAULT CHARACTER SET utf8mb4 
DEFAULT COLLATE utf8mb4_general_ci;

2) Créez dans cette base de données mediamarkt 3 tables (tous les champs non null) :

CREATE TABLE utilisateurs( 
	nom VARCHAR(100) COLLATE utf8mb4_general_ci NOT NULL, 
	prenom VARCHAR(100) COLLATE utf8mb4_general_ci NOT NULL, 
	age INT(11) NOT NULL, 
	date_de_naissance DATE NOT NULL, 
	adresse VARCHAR(250) COLLATE utf8mb4_general_ci NOT NULL, 
	email VARCHAR(100) COLLATE utf8mb4_general_ci NOT NULL 
) ENGINE=InnoDB;

CREATE TABLE tvs( 
	nom_TV VARCHAR(150) COLLATE utf8mb4_general_ci NOT NULL, 
	marque_TV VARCHAR(50) COLLATE utf8mb4_general_ci NOT NULL, 
	taille INT(20) NOT NULL 
) ENGINE=InnoDB;

CREATE TABLE smartphones( 
	nom_smartphone VARCHAR(150) COLLATE utf8mb4_general_ci NOT NULL, 
	marque_smartphone VARCHAR(50) COLLATE utf8mb4_general_ci NOT NULL,
	couleur VARCHAR(20) COLLATE utf8mb4_general_ci NOT NULL 
) ENGINE=InnoDB;

3) Modifier le champ couleur en un champ de type date dont le nom est « date_de_sortie ».

ALTER TABLE smartphones CHANGE couleur date_de_sortie DATE NOT NULL;

4) Rajouter pour ces 3 tables un champ id_nomdelatable qui sera la clé primaire en première position.

ALTER TABLE utilisateurs ADD id_utilisateurs INT AUTO_INCREMENT PRIMARY KEY FIRST;

ALTER TABLE tvs ADD id_tvs INT AUTO_INCREMENT PRIMARY KEY FIRST;

ALTER TABLE smartphones ADD id_smartphones INT AUTO_INCREMENT PRIMARY KEY FIRST;
